                    variation = random.uniform(0.9, 1.1)
                    words[i] = f"{original * variation:.2f}"
    
    # 2. Ajouter bruit aux bounding boxes (simule imprécisions OCR)
    if random.random() > 0.5:
        for i in range(len(boxes)):
            noise = [random.randint(-3, 3) for _ in range(4)]
            boxes[i] = [max(0, min(b + n, 1000)) for b, n in zip(boxes[i], noise)]
            
    return words, boxes, labels

4. Focal Loss pour gérer le déséquilibre des classes

class FocalLoss(nn.Module):
    def __init__(self, gamma=2.0):
        super(FocalLoss, self).__init__()
        self.gamma = gamma
        
    def forward(self, logits, target):
        # Calculer la focal loss qui donne plus de poids aux exemples difficiles
        ce_loss = F.cross_entropy(logits.view(-1, logits.size(-1)), target.view(-1), reduction='none')
        pt = torch.exp(-ce_loss)
        loss = ((1 - pt) ** self.gamma) * ce_loss
        return loss.mean()

# Utiliser dans l'entraînement
criterion = FocalLoss(gamma=2.0)
